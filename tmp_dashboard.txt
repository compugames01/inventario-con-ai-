import { lazy, Suspense, useState, useRef, useEffect, useMemo } from 'react';
import { useKV } from '@/hooks/use-electron-kv';
import { Product, User, AuditLog } from '@/lib/types';
import { Button } from '@/components/ui/button';
import { Badge } from '@/components/ui/badge';
import { Dialog, DialogContent, DialogDescription, DialogHeader, DialogTitle } from '@/components/ui/dialog';
import { AlertDialog, AlertDialogAction, AlertDialogCancel, AlertDialogContent, AlertDialogDescription, AlertDialogFooter, AlertDialogHeader, AlertDialogTitle } from '@/components/ui/alert-dialog';
import { Tabs, TabsContent, TabsList, TabsTrigger } from '@/components/ui/tabs';
import { StatsCard } from '@/components/StatsCard';
import { ProductForm } from '@/components/ProductForm';
import { InventoryTable } from '@/components/InventoryTable';
import { CriticalStockAlert } from '@/components/CriticalStockAlert';
import { JosimarLogo } from '@/components/JosimarLogo';
import { Plus, SignOut, Download, Package, Warning, CurrencyDollar, ShieldCheck, User as UserIcon, Database, Upload, ClockCounterClockwise, CheckCircle, Bell, Receipt, CalendarBlank, Users, LockKey, Moon, Sun, Trash, ShareNetwork } from '@phosphor-icons/react';
import { useTheme } from '@/components/theme-provider';
import { generateId, exportToCSV, formatCurrency, getStockStatus } from '@/lib/inventory-utils';
import { exportDatabase, importDatabase, createBackup } from '@/lib/database';
import { filterVisibleProducts, canEditProduct, canDeleteProduct, canAccessTab, getAccessibleTabs } from '@/lib/permissions-utils';
import { useAudit } from '@/hooks/use-audit';
import { toast } from 'sonner';
import { ScrollArea } from '@/components/ui/scroll-area';

interface DashboardProps {
  onLogout: () => void;
  currentUser: User | null;
}

type ViewType = 'inventory' | 'orders' | 'dailyclose' | 'users' | 'socials' | 'backup';

const getTabLabel = (tab: ViewType): string => {
  const labels: Record<ViewType, string> = {
    inventory: 'Inventario',
    orders: 'Ã“rdenes',
    dailyclose: 'Cierre del DÃ­a',
    users: 'Usuarios',
    socials: 'Mis Redes Sociales',
    backup: 'Base de Datos',
  };
  return labels[tab];
};

const OrdersPage = lazy(() => import('@/components/OrdersPage').then((m) => ({ default: m.OrdersPage })));
const DailyClose = lazy(() => import('@/components/DailyClose').then((m) => ({ default: m.DailyClose })));
const UserManagement = lazy(() => import('@/components/UserManagement').then((m) => ({ default: m.UserManagement })));
const SocialNetworks = lazy(() => import('@/components/SocialNetworks').then((m) => ({ default: m.SocialNetworks }))); 

export function Dashboard({ onLogout, currentUser }: DashboardProps) {
  const [currentView, setCurrentView] = useState<ViewType>('inventory');
  const [products, setProducts] = useKV<Product[]>('inventory-products', []);
  const [users, setUsers] = useKV<User[]>('system-users', []);
  const [dismissedAlerts, setDismissedAlerts] = useKV<string[]>('dismissed-stock-alerts', []);
  const [isAddDialogOpen, setIsAddDialogOpen] = useState(false);
  const [editingProduct, setEditingProduct] = useState<Product | null>(null);
  const [deletingProductId, setDeletingProductId] = useState<string | null>(null);
  const [showAuditDialog, setShowAuditDialog] = useState(false);
  const [showClearAuditConfirm, setShowClearAuditConfirm] = useState(false);
  const { theme, setTheme } = useTheme();
  const fileInputRef = useRef<HTMLInputElement>(null);
  
  const {
    auditLogs,
    logProductCreate,
    logProductUpdate,
    logProductDelete,
    logBackupRestore,
    clearAuditLogs
  } = useAudit();

  // Siempre leer permisos frescos desde system-users, no desde el currentUser cacheado al login
  const liveCurrentUser = useMemo(() => {
    if (!currentUser) return null;
    if (currentUser.isAdmin) return currentUser;
    const liveUser = (users || []).find(u => u.username === currentUser.username);
    return liveUser ?? currentUser;
  }, [currentUser, users]);

  const isAdmin = liveCurrentUser?.isAdmin ?? false;
  const safeProducts = products || [];
  const accessibleTabs = useMemo<ViewType[]>(() => {
    const baseTabs = getAccessibleTabs(liveCurrentUser ?? null).filter(t => !t.includes(':')) as Exclude<ViewType, 'backup'>[];
    const withSocials = baseTabs.includes('socials') ? baseTabs : [...baseTabs, 'socials'];
    return isAdmin ? [...withSocials, 'backup'] : withSocials;
  }, [liveCurrentUser, isAdmin]);
  const safeAccessibleTabs = accessibleTabs.length > 0 ? accessibleTabs : (['inventory'] as ViewType[]);
  const activeView = safeAccessibleTabs.includes(currentView) ? currentView : safeAccessibleTabs[0];
  
  const visibleProducts = useMemo(
    () => filterVisibleProducts(liveCurrentUser ?? null, safeProducts),
    [liveCurrentUser, safeProducts]
  );

  useEffect(() => {
    if (currentView !== activeView) {
      setCurrentView(activeView);
      toast.info(`Cambiado a ${getTabLabel(activeView)} - acceso restringido`);
    }
  }, [currentView, activeView]);

  useEffect(() => {
    const checkAutoBackup = async () => {
      const lastBackup = await window.spark.kv.get<string>('last-auto-backup');
      const now = Date.now();
      const dayInMs = 24 * 60 * 60 * 1000;

      if (!lastBackup || (now - new Date(lastBackup).getTime()) >= dayInMs) {
        if (safeProducts.length > 0) {
          const safeUsers = users || [];
          const backup = createBackup(safeProducts, safeUsers, auditLogs);
          exportDatabase(backup);
          await window.spark.kv.set('last-auto-backup', new Date().toISOString());
          toast.success('Backup automÃ¡tico de la base de datos realizado');
        }
      }
    };

    if (isAdmin) {
      checkAutoBackup();
    }
  }, [isAdmin, safeProducts.length, users, auditLogs]);

  const handleAddProduct = (data: {
    sku: string;
    name: string;
    category: string;
    quantity: number;
    price: number;
    supplier: string;
    description?: string;
  }) => {
    const newProduct: Product = {
      ...data,
      id: generateId(),
      createdAt: new Date().toISOString(),
      updatedAt: new Date().toISOString(),
    };

    setProducts((current) => [...(current || []), newProduct]);
    logProductCreate(newProduct);
    setIsAddDialogOpen(false);
    toast.success('Producto agregado y registrado en auditorÃ­a');
  };

  const handleEditProduct = (data: {
    sku: string;
    name: string;
    category: string;
    quantity: number;
    price: number;
    supplier: string;
    description?: string;
  }) => {
    if (!editingProduct) return;

    const oldProduct = editingProduct;
    const updatedProduct = { ...oldProduct, ...data, updatedAt: new Date().toISOString() };

    setProducts((current) =>
      (current || []).map((p) =>
        p.id === editingProduct.id ? updatedProduct : p
      )
    );
    
    logProductUpdate(oldProduct, updatedProduct);
    setEditingProduct(null);
    toast.success('Producto actualizado y cambios registrados');
  };

  const handleDeleteProduct = async () => {
    if (!deletingProductId) return;

    const productToDelete = safeProducts.find((p) => p.id === deletingProductId);
    if (productToDelete) {
      logProductDelete(productToDelete);
    }

    setProducts((current) => (current || []).filter((p) => p.id !== deletingProductId));
    
    try {
      const closeHistory = await window.spark.kv.get<any[]>('daily-close-history');
      if (closeHistory && closeHistory.length > 0) {
        const updatedHistory = closeHistory.map(report => {
          if (!report.topProducts || report.topProducts.length === 0) {
            return report;
          }

          const updatedTopProducts = report.topProducts.filter(
            (p: any) => p.productId !== deletingProductId
          );
          
          const removedProduct = report.topProducts.find(
            (p: any) => p.productId === deletingProductId
          );
          
          let updatedFinancial = { ...report.financial };
          
          if (removedProduct) {
            const revenueToSubtract = removedProduct.revenue || 0;
            updatedFinancial = {
              ...report.financial,
              grossRevenue: Math.max(0, report.financial.grossRevenue - revenueToSubtract),
              netProfit: Math.max(0, report.financial.netProfit - revenueToSubtract),
            };
            
            if (updatedFinancial.grossRevenue > 0) {
              updatedFinancial.profitMargin = 
                (updatedFinancial.netProfit / updatedFinancial.grossRevenue) * 100;
            } else {
              updatedFinancial.profitMargin = 0;
            }
          }
          
          return {
            ...report,
            topProducts: updatedTopProducts,
            financial: updatedFinancial,
          };
        });
        
        await window.spark.kv.set('daily-close-history', updatedHistory);
      }
    } catch (error) {
      console.error('Error al actualizar historial de cierres:', error);
    }
    
    setDeletingProductId(null);
    toast.success('Producto eliminado del inventario y del historial de cierres');
  };

  const handleExport = () => {
    if (safeProducts.length === 0) {
      toast.error('No hay productos para exportar');
      return;
    }
    exportToCSV(safeProducts);
    toast.success('Inventario exportado a CSV exitosamente');
  };

  const handleBackupDatabase = () => {
    const safeUsers = users || [];
    const backup = createBackup(safeProducts, safeUsers, auditLogs);
    exportDatabase(backup);
    toast.success('Base de datos guardada correctamente');
  };

  const handleRestoreDatabase = async (event: React.ChangeEvent<HTMLInputElement>) => {
    const file = event.target.files?.[0];
    if (!file) return;
    try {
      const backup = await importDatabase(file);
      setProducts(backup.products || []);
      setUsers(backup.users || []);
      logBackupRestore(file.name);
      toast.success(`Base de datos restaurada: ${backup.metadata?.productsCount || 0} productos y ${backup.metadata?.usersCount || 0} usuarios`);
    } catch (error) {
      console.error('Error al restaurar base de datos:', error);
      toast.error('Error al restaurar la base de datos. Verifique el archivo.');
    }
    event.target.value = '';
  };

  const handleDismissAlert = (sku: string) => {
    setDismissedAlerts((current) => [...(current || []), sku]);
  };

  const totalProducts = visibleProducts.length;
  const lowStockProducts = visibleProducts.filter((p) => {
    const status = getStockStatus(p.quantity);
    return status === 'low-stock' || status === 'out-of-stock';
  }).length;
  const totalValue = visibleProducts.reduce((sum, p) => sum + p.price * p.quantity, 0);
  const activeCriticalAlerts = visibleProducts.filter(
    (p) => getStockStatus(p.quantity) !== 'in-stock' && !(dismissedAlerts || []).includes(p.sku)
  ).length;

  return (
    <div className="flex flex-col min-h-screen bg-background">
      <header className="sticky top-0 z-50 w-full border-b-2 border-border/30 bg-card/80 backdrop-blur-xl shadow-lg">
        <div className="px-3 sm:px-6 lg:px-8 h-16 sm:h-18 lg:h-20 flex items-center">
          <div className="flex items-center justify-between gap-2 sm:gap-4 w-full">
            <div className="flex items-center gap-2 sm:gap-3 lg:gap-5 min-w-0 flex-1">
              <div className="relative group shrink-0">
                <JosimarLogo size="clamp(40px, 8vw, 64px)" />
              </div>
              <div className="flex flex-col min-w-0">
                <h1 className="text-base sm:text-xl lg:text-2xl xl:text-3xl font-bold tracking-tight bg-gradient-to-r from-primary via-accent to-primary bg-clip-text text-transparent truncate">
                  Josimar Cell
                </h1>
              </div>
            </div>
            <div className="flex items-center gap-1.5 sm:gap-2 lg:gap-4 shrink-0">
              {activeCriticalAlerts > 0 && currentView === 'inventory' && (
                <div className="relative">
                  <Button variant="outline" size="icon" className="relative h-9 w-9 sm:h-10 sm:w-10 lg:h-12 lg:w-12 border-2 border-warning/30 hover:bg-warning/10 hover:border-warning transition-all duration-200 shadow-lg shadow-warning/10">
                    <Bell className="w-4 sm:w-5 h-4 sm:h-5 text-warning" weight="fill" />
                    <span className="absolute -top-1.5 -right-1.5 sm:-top-2 sm:-right-2 flex h-5 w-5 sm:h-6 sm:w-6 items-center justify-center rounded-full bg-gradient-to-br from-destructive to-destructive/80 text-white text-xs font-bold shadow-lg animate-pulse">
                      {activeCriticalAlerts > 9 ? '9+' : activeCriticalAlerts}
                    </span>
                  </Button>
                </div>
              )}
              <Button
                variant="ghost"
                size="icon"
                onClick={() => setTheme(theme === 'dark' ? 'light' : 'dark')}
                className="h-9 w-9 sm:h-10 sm:w-10 lg:h-12 lg:w-12 hover:bg-muted/80 transition-all"
              >
                {theme === 'dark' ? <Sun className="w-4 sm:w-5 h-4 sm:h-5" weight="duotone" /> : <Moon className="w-4 sm:w-5 h-4 sm:h-5" weight="duotone" />}
              </Button>
              {liveCurrentUser && (
                <div className="hidden sm:flex items-center gap-2 lg:gap-3 px-3 lg:px-4 py-2 lg:py-2.5 rounded-xl bg-muted/50 border border-border/50">
                  <div className="w-7 h-7 sm:w-8 sm:h-8 lg:w-9 lg:h-9 rounded-lg bg-gradient-to-br from-primary/20 to-primary/10 flex items-center justify-center">
                    {liveCurrentUser.isAdmin ? <ShieldCheck className="w-4 sm:w-4.5 lg:w-5 h-4 sm:h-4.5 lg:h-5 text-primary" weight="fill" /> : <UserIcon className="w-4 sm:w-4.5 lg:w-5 h-4 sm:h-4.5 lg:h-5 text-muted-foreground" />}
                  </div>
                  <span className="text-sm lg:text-base font-bold text-foreground truncate max-w-[100px] lg:max-w-none">{liveCurrentUser.username}</span>
                </div>
              )}
              <Button
                variant="outline"
                size="sm"
                onClick={onLogout}
                className="h-9 sm:h-10 lg:h-12 px-3 sm:px-4 lg:px-5 border-2 hover:bg-destructive/10 hover:border-destructive/50 hover:text-destructive transition-all duration-200 font-semibold text-xs sm:text-sm shadow-sm"
              >
                <SignOut className="w-3.5 sm:w-4 lg:w-5 h-3.5 sm:h-4 lg:h-5 sm:mr-1.5 lg:mr-2" weight="bold" />
                <span className="hidden sm:inline">Salir</span>
              </Button>
            </div>
          </div>
        </div>
      </header>

      <main className="flex-1 overflow-y-auto px-3 sm:px-6 lg:px-8 py-4 sm:py-6 lg:py-8">
        <Tabs value={activeView} onValueChange={(value) => {
          if (safeAccessibleTabs.includes(value as ViewType)) setCurrentView(value as ViewType);
        }} className="w-full">
          <TabsList className="grid w-full max-w-5xl mx-auto mb-6 sm:mb-8 h-auto sm:h-14 bg-background/60 dark:bg-muted/40 backdrop-blur-sm p-1 sm:p-1.5 shadow-lg border-2 border-border/30 rounded-xl sm:rounded-2xl" style={{ gridTemplateColumns: `repeat(${safeAccessibleTabs.length}, 1fr)` }}>
            {canAccessTab(liveCurrentUser ?? null, 'inventory') && (
              <TabsTrigger value="inventory" className="data-[state=active]:bg-gradient-to-br data-[state=active]:from-primary data-[state=active]:to-primary/90 data-[state=active]:text-white data-[state=active]:shadow-xl transition-all duration-300 font-bold text-xs sm:text-sm rounded-lg sm:rounded-xl py-2 sm:py-2.5 px-2 sm:px-3">
                <Package className="w-4 sm:w-5 h-4 sm:h-5 sm:mr-1.5" weight="duotone" />
                <span className="hidden sm:inline">Inventario</span>
              </TabsTrigger>
            )}
            {canAccessTab(liveCurrentUser ?? null, 'orders') && (
              <TabsTrigger value="orders" className="data-[state=active]:bg-gradient-to-br data-[state=active]:from-accent data-[state=active]:to-accent/90 data-[state=active]:text-white data-[state=active]:shadow-xl transition-all duration-300 font-bold text-xs sm:text-sm rounded-lg sm:rounded-xl py-2 sm:py-2.5 px-2 sm:px-3">
                <Receipt className="w-4 sm:w-5 h-4 sm:h-5 sm:mr-1.5" weight="duotone" />
                <span className="hidden sm:inline">Ã“rdenes</span>
              </TabsTrigger>
            )}
            {canAccessTab(liveCurrentUser ?? null, 'dailyclose') && (
              <TabsTrigger value="dailyclose" className="data-[state=active]:bg-gradient-to-br data-[state=active]:from-success data-[state=active]:to-success/90 data-[state=active]:text-white data-[state=active]:shadow-xl transition-all duration-300 font-bold text-xs sm:text-sm rounded-lg sm:rounded-xl py-2 sm:py-2.5 px-2 sm:px-3">
                <CalendarBlank className="w-4 sm:w-5 h-4 sm:h-5 sm:mr-1.5" weight="duotone" />
                <span className="hidden sm:inline">Cierre del DÃ­a</span>
              </TabsTrigger>
            )}
            {canAccessTab(liveCurrentUser ?? null, 'users') && (
              <TabsTrigger value="users" className="data-[state=active]:bg-gradient-to-br data-[state=active]:from-warning data-[state=active]:to-warning/90 data-[state=active]:text-warning-foreground data-[state=active]:shadow-xl transition-all duration-300 font-bold text-xs sm:text-sm rounded-lg sm:rounded-xl py-2 sm:py-2.5 px-2 sm:px-3">
                <Users className="w-4 sm:w-5 h-4 sm:h-5 sm:mr-1.5" weight="duotone" />
                <span className="hidden sm:inline">Usuarios</span>
              </TabsTrigger>
            )}
            {isAdmin && safeAccessibleTabs.includes('backup') && (
              <TabsTrigger value="backup" className="data-[state=active]:bg-gradient-to-br data-[state=active]:from-primary data-[state=active]:to-accent data-[state=active]:text-white data-[state=active]:shadow-xl transition-all duration-300 font-bold text-xs sm:text-sm rounded-lg sm:rounded-xl py-2 sm:py-2.5 px-2 sm:px-3">
                <Database className="w-4 sm:w-5 h-4 sm:h-5 sm:mr-1.5" weight="duotone" />
                <span className="hidden sm:inline">Base de Datos</span>
              </TabsTrigger>
            )}
            {safeAccessibleTabs.includes('socials') && (
              <TabsTrigger value="socials" className="data-[state=active]:bg-gradient-to-br data-[state=active]:from-primary data-[state=active]:to-primary/90 data-[state=active]:text-white data-[state=active]:shadow-xl transition-all duration-300 font-bold text-xs sm:text-sm rounded-lg sm:rounded-xl py-2 sm:py-2.5 px-2 sm:px-3">
                <ShareNetwork className="w-4 sm:w-5 h-4 sm:h-5 sm:mr-1.5" weight="duotone" />
                <span className="hidden sm:inline">Redes Sociales</span>
                <span className="sm:hidden">Social</span>
              </TabsTrigger>
            )}
          </TabsList>

          <TabsContent value="inventory" className="space-y-6 sm:space-y-8 animate-slide-up">
            <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-3 gap-4 sm:gap-6">
              <StatsCard title="Total de Productos" value={totalProducts} icon={Package} iconColor="text-primary" />
              <StatsCard title="Stock Bajo / Agotado" value={lowStockProducts} icon={Warning} iconColor="text-warning" />
              <StatsCard title="Valor Total del Inventario" value={formatCurrency(totalValue)} icon={CurrencyDollar} iconColor="text-success" />
            </div>

            {!isAdmin && safeProducts.length > 0 && (
              <div className="p-4 sm:p-5 bg-gradient-to-br from-accent/10 to-accent/5 border-2 border-accent/30 rounded-xl shadow-md">
                <div className="flex items-start gap-3 sm:gap-4">
                  <div className="w-10 h-10 rounded-lg bg-accent/20 flex items-center justify-center shrink-0">
                    <LockKey className="w-5 h-5 text-accent" weight="duotone" />
                  </div>
                  <div className="space-y-1">
                    <p className="text-sm font-bold text-accent-foreground">Permisos Granulares Activos</p>
                    <p className="text-xs text-muted-foreground">
                      Puede ver {visibleProducts.length} de {safeProducts.length} productos.
                    </p>
                  </div>
                </div>
              </div>
            )}

            <CriticalStockAlert products={visibleProducts} onDismiss={handleDismissAlert} dismissedAlerts={dismissedAlerts || []} />

            <div className="bg-card rounded-xl sm:rounded-2xl shadow-lg border-2 border-border/30 overflow-hidden">
              <div className="px-4 sm:px-6 lg:px-8 py-4 sm:py-5 bg-gradient-to-r from-card to-secondary/20 border-b-2 border-border/30">
                <div className="flex flex-col lg:flex-row gap-3 sm:gap-4 justify-between items-start">
                  <div>
                    <h2 className="text-lg sm:text-xl lg:text-2xl font-bold text-foreground">Productos</h2>
                  </div>
                  <div className="flex gap-2 sm:gap-3 flex-wrap w-full lg:w-auto">
                    {isAdmin && (
                      <Button variant="outline" onClick={() => setShowAuditDialog(true)} size="sm" className="border-2 hover:bg-accent/5 hover:border-accent/50 transition-all font-semibold text-xs sm:text-sm">
                        <ClockCounterClockwise className="w-3 h-3 sm:w-4 sm:h-4 sm:mr-2" weight="duotone" />
                        <span className="hidden sm:inline">AuditorÃ­a ({auditLogs.length})</span>
                        <span className="sm:hidden">({auditLogs.length})</span>
                      </Button>
                    )}
                    <Button variant="outline" onClick={handleExport} disabled={visibleProducts.length === 0} size="sm" className="border-2 hover:bg-accent/5 hover:border-accent/50 transition-all font-semibold text-xs sm:text-sm">
                      <Download className="w-3 h-3 sm:w-4 sm:h-4 sm:mr-2" weight="duotone" />
                      <span className="hidden sm:inline">Exportar CSV</span>
                      <span className="sm:hidden">CSV</span>
                    </Button>
                    {isAdmin && (
                      <Button onClick={() => setIsAddDialogOpen(true)} size="sm" className="bg-gradient-to-br from-primary to-primary/90 hover:from-primary/90 hover:to-primary shadow-lg hover:shadow-xl transition-all font-bold text-xs sm:text-sm">
                        <Plus className="w-4 h-4 sm:w-5 sm:h-5 sm:mr-2" weight="bold" />
                        <span className="hidden sm:inline">Agregar Producto</span>
                        <span className="sm:hidden">+</span>
                      </Button>
                    )}
                  </div>
                </div>
              </div>
              <div className="p-3 sm:p-4 lg:p-6">
                <InventoryTable
                  products={visibleProducts}
                  currentUser={liveCurrentUser}
                  onEdit={
                    isAdmin ? setEditingProduct
                    : liveCurrentUser ? (product) => {
                        if (canEditProduct(liveCurrentUser, product)) setEditingProduct(product);
                        else toast.error('No tiene permisos para editar este producto');
                      }
                    : undefined
                  }
                  onDelete={
                    isAdmin ? setDeletingProductId
                    : liveCurrentUser ? (productId) => {
                        const product = safeProducts.find((p) => p.id === productId);
                        if (product && canDeleteProduct(liveCurrentUser, product)) setDeletingProductId(productId);
                        else toast.error('No tiene permisos para eliminar este producto');
                      }
                    : undefined
                  }
                />
              </div>
            </div>
          </TabsContent>

          <TabsContent value="orders">
            {liveCurrentUser && (
              <Suspense fallback={<div className='py-6 text-sm text-muted-foreground'>Cargando Ã³rdenes...</div>}>
                <OrdersPage products={safeProducts} currentUser={liveCurrentUser} onUpdateProducts={(updater) => setProducts((current) => updater(current || []))} />
              </Suspense>
            )}
          </TabsContent>

          <TabsContent value="dailyclose">
            {liveCurrentUser && (
              <Suspense fallback={<div className='py-6 text-sm text-muted-foreground'>Cargando cierre...</div>}>
                <DailyClose products={safeProducts} currentUser={liveCurrentUser} />
              </Suspense>
            )}
          </TabsContent>

          <TabsContent value="users">
            {liveCurrentUser && isAdmin && (
              <Suspense fallback={<div className='py-6 text-sm text-muted-foreground'>Cargando usuarios...</div>}>
                <UserManagement currentUser={liveCurrentUser} />
              </Suspense>
            )}
          </TabsContent>

          <TabsContent value="backup">
            {isAdmin && (
              <div className="max-w-4xl mx-auto">
                <div className="bg-card rounded-xl sm:rounded-2xl shadow-lg border-2 border-border/30 overflow-hidden">
                  <div className="px-4 sm:px-6 lg:px-8 py-4 sm:py-5 bg-gradient-to-r from-card to-secondary/20 border-b-2 border-border/30">
                    <h2 className="text-lg sm:text-xl lg:text-2xl font-bold text-foreground">Respaldo de Base de Datos</h2>
                    <p className="text-sm text-muted-foreground mt-1">Guarde o cargue la informaciÃ³n del sistema desde un archivo.</p>
                  </div>
                  <div className="p-4 sm:p-6 lg:p-8">
                    <div className="grid grid-cols-1 md:grid-cols-2 gap-4">
                      <Button
                        variant="outline"
                        onClick={handleBackupDatabase}
                        className="h-14 sm:h-16 border-2 hover:bg-primary/5 hover:border-primary/50 transition-all font-semibold text-sm sm:text-base"
                      >
                        <Database className="w-5 h-5 mr-2" weight="duotone" />
                        Guardar BD
                      </Button>
                      <Button
                        variant="outline"
                        onClick={() => fileInputRef.current?.click()}
                        className="h-14 sm:h-16 border-2 hover:bg-success/5 hover:border-success/50 transition-all font-semibold text-sm sm:text-base"
                      >
                        <Upload className="w-5 h-5 mr-2" weight="duotone" />
                        Cargar BD
                      </Button>
                      <input ref={fileInputRef} type="file" accept=".json,.sib" onChange={handleRestoreDatabase} className="hidden" />
                    </div>
                  </div>
                </div>
              </div>
            )}
          </TabsContent>

          <TabsContent value="socials">
            <Suspense fallback={<div className='py-6 text-sm text-muted-foreground'>Cargando redes sociales...</div>}>
              <SocialNetworks />
            </Suspense>
          </TabsContent>

        </Tabs>
      </main>

      {isAdmin && (
        <>
          <Dialog open={isAddDialogOpen} onOpenChange={setIsAddDialogOpen}>
            <DialogContent className="max-w-4xl w-[95vw] h-[95vh] sm:h-[85vh] flex flex-col p-0 gap-0 overflow-hidden">
              <ProductForm
                onSubmit={handleAddProduct}
                onCancel={() => setIsAddDialogOpen(false)}
                existingSkus={safeProducts.map((p) => p.sku)}
              />
            </DialogContent>
          </Dialog>

          <Dialog open={!!editingProduct} onOpenChange={(open) => { if (!open) setEditingProduct(null); }}>

                  product={editingProduct}
                  onSubmit={handleEditProduct}
                  onCancel={() => setEditingProduct(null)}
                  existingSkus={safeProducts.map((p) => p.sku)}
                />
              )}
            </DialogContent>
          </Dialog>
        </>
      )}

      {!isAdmin && editingProduct && (
        <Dialog open={!!editingProduct} onOpenChange={(open) => { if (!open) setEditingProduct(null); }}>
          <DialogContent className="max-w-4xl w-[95vw] h-[95vh] sm:h-[85vh] flex flex-col p-0 gap-0 overflow-hidden">
            <ProductForm
              product={editingProduct}
              onSubmit={handleEditProduct}
              onCancel={() => setEditingProduct(null)}
              existingSkus={safeProducts.map((p) => p.sku)}
            />
          </DialogContent>
        </Dialog>

        <AlertDialogContent>

            <AlertDialogTitle>Â¿Eliminar producto?</AlertDialogTitle>
            <AlertDialogDescription>
              Esta acciÃ³n no se puede deshacer. El producto serÃ¡ eliminado permanentemente del inventario.
            </AlertDialogDescription>
          </AlertDialogHeader>
          <AlertDialogFooter>
            <AlertDialogCancel>Cancelar</AlertDialogCancel>
            <AlertDialogAction onClick={handleDeleteProduct} className="bg-destructive hover:bg-destructive/90">
              <Trash className="w-4 h-4 mr-2" />
              Eliminar
            </AlertDialogAction>
          </AlertDialogFooter>
        </AlertDialogContent>
      </AlertDialog>

      <Dialog open={showAuditDialog} onOpenChange={setShowAuditDialog}>
        <DialogContent className="max-w-4xl w-[95vw] h-[90vh] flex flex-col p-0 gap-0 overflow-hidden">
          <DialogHeader className="px-6 py-4 border-b shrink-0">
            <DialogTitle className="flex items-center gap-2">
              <ClockCounterClockwise className="w-5 h-5 text-accent" weight="duotone" />
              Historial de AuditorÃ­a
            </DialogTitle>
            <DialogDescription>Historial completo de cambios en el inventario</DialogDescription>
          </DialogHeader>
          <ScrollArea className="flex-1 px-6 py-4">
            {auditLogs.length === 0 ? (
              <p className="text-center text-muted-foreground py-8">No hay registros de auditorÃ­a</p>
            ) : (
              <div className="space-y-3">
                {[...auditLogs].reverse().map((log) => (
                  <div key={log.id} className="p-3 rounded-lg border bg-card/50">
                    <div className="flex items-center justify-between gap-2 mb-1">
                      <div className="flex items-center gap-2">
                        <Badge variant={log.action === 'CREATE' ? 'default' : log.action === 'DELETE' ? 'destructive' : 'secondary'} className="text-xs">
                          {log.action}
                        </Badge>
                        <span className="text-sm font-medium">{log.entityName}</span>
                      </div>
                      <span className="text-xs text-muted-foreground">{new Date(log.timestamp).toLocaleString('es-ES')}</span>
                    </div>
                    <p className="text-xs text-muted-foreground">Por: {log.username}</p>
                    {log.changes && log.changes.length > 0 && (
                      <div className="mt-2 space-y-1">
                        {log.changes.map((change, i) => (
                          <p key={i} className="text-xs text-muted-foreground">
                            <span className="font-medium">{change.field}:</span> {String(change.oldValue)} â†’ {String(change.newValue)}
                          </p>
                        ))}
                      </div>
                    )}
                  </div>
                ))}
              </div>
            )}
          </ScrollArea>
          {isAdmin && auditLogs.length > 0 && (
            <div className="px-6 py-4 border-t shrink-0">
              <Button variant="outline" size="sm" onClick={() => setShowClearAuditConfirm(true)} className="text-destructive hover:bg-destructive/10">
                Limpiar historial
              </Button>
            </div>
          )}
        </DialogContent>
      </Dialog>
      <AlertDialog open={showClearAuditConfirm} onOpenChange={setShowClearAuditConfirm}>
        <AlertDialogContent>
          <AlertDialogHeader>
            <AlertDialogTitle>¿Estás seguro?</AlertDialogTitle>
            <AlertDialogDescription>
              Esta acción eliminará todo el historial de auditoría y no se puede deshacer.
            </AlertDialogDescription>
          </AlertDialogHeader>
          <AlertDialogFooter>
            <AlertDialogCancel>No</AlertDialogCancel>
            <AlertDialogAction
              className="bg-destructive hover:bg-destructive/90"
              onClick={async () => {
                const ok = await clearAuditLogs();
                setShowClearAuditConfirm(false);
                if (ok) {
                  toast.success('Historial de auditoría eliminado correctamente');
                  setShowAuditDialog(false);
                } else {
                  toast.error('No se pudo eliminar el historial de auditoría');
                }
              }}
            >
              Sí
            </AlertDialogAction>
          </AlertDialogFooter>
        </AlertDialogContent>
      </AlertDialog>
    </div>
  );
}
